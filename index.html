/* ========================================================================== SISTEMA DE MOVIMIENTO FLUIDO Y COLISIÓN POR DESLIZAMIENTO (60 FPS) ========================================================================== */ /** * 1. Inicialización en tu método de inicio / reinicio del juego */ initSmoothControls(startX, startY, goalCol, goalRow) { this.isDragging = false; this.isGameFinished = false; // Radio circular estricto del trébol (evita atravesar cualquier bloque 0) this.playerRadius = Math.round(this.tileSize * 0.28); // Posición continua objetivo en píxeles (donde se ubica el dedo) this.targetTouchX = startX; this.targetTouchY = startY; this.goalGridCol = goalCol; this.goalGridRow = goalRow; // Iniciar bucle de renderizado a 60fps con requestAnimationFrame this.startSmoothLoop(); } /** * 2. Comprobación matemática continua contra la matriz (0 = Muro, 1 = Camino) * Verifica si el círculo del trébol intersecta con cualquier bloque sólido circundante. */ isCollidingWithWalls(x, y, radius) { const minCol = Math.floor((x - radius - this.offsetX) / this.tileSize); const maxCol = Math.floor((x + radius - this.offsetX) / this.tileSize); const minRow = Math.floor((y - radius - this.offsetY) / this.tileSize); const maxRow = Math.floor((y + radius - this.offsetY) / this.tileSize); for (let r = minRow; r <= maxRow; r++) { for (let c = minCol; c <= maxCol; c++) { // Si la casilla está fuera del mapa o es 0 (muro sólido) const isSolid = (r < 0 || r >= this.mazeHeight || c < 0 || c >= this.mazeWidth) || (this.mazeGrid[r][c] === 0); if (isSolid) { const wx1 = this.offsetX + c * this.tileSize; const wy1 = this.offsetY + r * this.tileSize; const wx2 = wx1 + this.tileSize; const wy2 = wy1 + this.tileSize; // Punto más cercano de la caja del muro al centro del trébol const closestX = Math.max(wx1, Math.min(x, wx2)); const closestY = Math.max(wy1, Math.min(y, wy2)); const diffX = x - closestX; const diffY = y - closestY; // Si la distancia al muro es menor que el radio del trébol, hay colisión if ((diffX * diffX + diffY * diffY) < (radius * radius)) { return true; } } } } return false; } /** * 3. Actualización de movimiento con Colisión por Deslizamiento (Sliding Collision): * Descompone el vector de movimiento en X e Y. Si un eje colisiona, el otro continúa, * permitiendo doblar esquinas y resbalar por las paredes con total suavidad. */ stepSmoothMovement() { if (!this.isDragging || this.isGameFinished || !this.player) return; const currentX = this.player.x; const currentY = this.player.y; const dx = this.targetTouchX - currentX; const dy = this.targetTouchY - currentY; const dist = Math.hypot(dx, dy); if (dist < 0.5) return; // Suavizado continuo (lerp responsivo para pantallas táctiles) const lerpFactor = 0.32; const moveDist = Math.min(dist, Math.max(dist * lerpFactor, 1.5)); const angle = Math.atan2(dy, dx); const deltaX = Math.cos(angle) * moveDist; const deltaY = Math.sin(angle) * moveDist; // Sub-pasos en píxeles (máx 2px) para evitar saltos o que traspase esquinas const maxStepSize = 2.0; const steps = Math.ceil(Math.max(Math.abs(deltaX), Math.abs(deltaY)) / maxStepSize) || 1; const stepX = deltaX / steps; const stepY = deltaY / steps; let posX = currentX; let posY = currentY; const r = this.playerRadius; for (let i = 0; i < steps; i++) { // A. DESLIZAMIENTO EN EJE X: si choca con muro en X, detiene X pero permite Y const nextX = posX + stepX; if (!this.isCollidingWithWalls(nextX, posY, r)) { posX = nextX; } else { // Asistencia en esquinas: ligero desvío hacia el centro del pasillo en Y const cellCenterY = this.offsetY + Math.floor((posY - this.offsetY) / this.tileSize) * this.tileSize + this.tileSize / 2; const nudgeY = (cellCenterY - posY) * 0.18; if (Math.abs(nudgeY) > 0.2 && !this.isCollidingWithWalls(posX, posY + nudgeY, r)) { posY += nudgeY; } } // B. DESLIZAMIENTO EN EJE Y: si choca con muro en Y, detiene Y pero permite X const nextY = posY + stepY; if (!this.isCollidingWithWalls(posX, nextY, r)) { posY = nextY; } else { // Asistencia en esquinas: ligero desvío hacia el centro del pasillo en X const cellCenterX = this.offsetX + Math.floor((posX - this.offsetX) / this.tileSize) * this.tileSize + this.tileSize / 2; const nudgeX = (cellCenterX - posX) * 0.18; if (Math.abs(nudgeX) > 0.2 && !this.isCollidingWithWalls(posX + nudgeX, posY, r)) { posX += nudgeX; } } } // Actualizar posición continua en píxeles del trébol this.player.setPosition(posX, posY); if (this.playerAura) { this.playerAura.setPosition(posX, posY); } // Detección fluida de llegada a la meta const goalCenterX = this.offsetX + this.goalGridCol * this.tileSize + this.tileSize / 2; const goalCenterY = this.offsetY + this.goalGridRow * this.tileSize + this.tileSize / 2; if (Math.hypot(posX - goalCenterX, posY - goalCenterY) < this.tileSize * 0.52) { this.handleVictory(); } } /** * 4. Bucle a 60fps sincronizado mediante requestAnimationFrame */ startSmoothLoop() { const loop = () => { if (!this.isGameFinished) { this.stepSmoothMovement(); } this.rafId = requestAnimationFrame(loop); }; if (this.rafId) { cancelAnimationFrame(this.rafId); } this.rafId = requestAnimationFrame(loop); } stopSmoothLoop() { if (this.rafId) { cancelAnimationFrame(this.rafId); this.rafId = null; } } /** * 5. Eventos Touch con preventDefault() y captura continua de coordenadas */ setupTouchControls() { const canvas = this.game.canvas; if (canvas) { canvas.addEventListener('touchstart', (e) => { e.preventDefault(); }, { passive: false }); canvas.addEventListener('touchmove', (e) => { e.preventDefault(); if (this.isDragging && e.touches.length > 0 && !this.isGameFinished) { const rect = canvas.getBoundingClientRect(); const scaleX = canvas.width / rect.width; const scaleY = canvas.height / rect.height; this.targetTouchX = (e.touches[0].clientX - rect.left) * scaleX; this.targetTouchY = (e.touches[0].clientY - rect.top) * scaleY; } }, { passive: false }); canvas.addEventListener('touchend', (e) => { e.preventDefault(); this.stopDrag(); }, { passive: false }); canvas.addEventListener('touchcancel', (e) => { e.preventDefault(); this.stopDrag(); }, { passive: false }); } // Soporte de pointer para emulación en PC o ratón this.input.on('pointerdown', (pointer) => { if (this.isGameFinished) return; if (pointer.event && pointer.event.preventDefault) { pointer.event.preventDefault(); } const dist = Phaser.Math.Distance.Between(pointer.x, pointer.y, this.player.x, this.player.y); const grabTolerance = this.tileSize * 1.45; if (dist <= grabTolerance) { this.isDragging = true; this.targetTouchX = pointer.x; this.targetTouchY = pointer.y; this.tweens.add({ targets: [this.player, this.playerAura], scale: 1.22, duration: 120, ease: 'Back.easeOut' }); } }); this.input.on('pointermove', (pointer) => { if (this.isDragging && !this.isGameFinished) { if (pointer.event && pointer.event.preventDefault) { pointer.event.preventDefault(); } this.targetTouchX = pointer.x; this.targetTouchY = pointer.y; } }); this.input.on('pointerup', () => this.stopDrag()); this.input.on('pointerupoutside', () => this.stopDrag()); } stopDrag() { if (this.isDragging) { this.isDragging = false; this.tweens.add({ targets: [this.player, this.playerAura], scale: 1.0, duration: 120, ease: 'Sine.easeOut' }); } }
